declare
     i number(30):=1;
     n number(30):=5;
     f number(30):=1;

BEGIN
FOR i in 1..n LOOP
      f:=f*i;
      DBMS_OUTPUT.PUT_LINE('FACTORIAL OF '||i||' IS:- '||f);
   END LOOP;
end;

declare
     i number(30):=1;
     n number(30):=10;
    
BEGIN
FOR i in reverse 1..n LOOP

      DBMS_OUTPUT.PUt(i||' ');
      if mod(i,2)=0 then
       DBMS_OUTPUT.PUT_LINE('even');
      else
       DBMS_OUTPUT.PUT_LINE('odd');
      end if;
   END LOOP;
end;
